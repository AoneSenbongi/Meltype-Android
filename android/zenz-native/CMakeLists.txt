cmake_minimum_required(VERSION 3.22)
project(meltype_zenz LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to azooKey/llama.cpp commit 10131b23ee0dbd9d03dfc618cec9fe3402919277")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(meltype_zenz SHARED meltype_zenz.cpp)
target_link_libraries(meltype_zenz PRIVATE llama)
target_include_directories(meltype_zenz PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(meltype_zenz PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
  target_link_options(meltype_zenz PRIVATE "-Wl,-z,max-page-size=16384")
endif()
add_executable(zenz-smoke smoke.cpp)
target_link_libraries(zenz-smoke PRIVATE meltype_zenz)
if(ANDROID)
  target_link_options(zenz-smoke PRIVATE "-Wl,-z,max-page-size=16384")
endif()
